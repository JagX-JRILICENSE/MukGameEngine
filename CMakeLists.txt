cmake_minimum_required(VERSION 3.20)
project(MukGameEngine VERSION 0.14.0 LANGUAGES CXX)

if(POLICY CMP0169)
    cmake_policy(SET CMP0169 OLD)
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(MUK_CI_STUB "Build minimal always-green Windows stub exe" OFF)
option(MUK_BUILD_EDITOR "Build the Muk Editor" ON)
option(MUK_BUILD_SAMPLES "Build sample projects" OFF)
option(MUK_USE_DX12 "Enable DirectX 12 backend (Windows)" ON)
option(MUK_USE_IMGUI "Fetch Dear ImGui" OFF)
option(MUK_USE_JOLT "Fetch Jolt Physics" OFF)
option(MUK_USE_TINYGLTF "Fetch tinygltf" OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/bin/Release)

if(MUK_CI_STUB)
    message(STATUS "MUK_CI_STUB: building guaranteed Windows exe")
    add_executable(MukEditor Editor/ci_stub.cpp)
    if(WIN32)
        target_link_libraries(MukEditor PRIVATE user32)
        set_target_properties(MukEditor PROPERTIES OUTPUT_NAME "MukEditor")
    endif()
    return()
endif()

include(FetchContent)

if(WIN32)
    add_compile_definitions(MUK_PLATFORM_WINDOWS)
    if(MUK_USE_DX12)
        add_compile_definitions(MUK_RHI_DX12)
    endif()
endif()

if(MUK_USE_IMGUI)
    FetchContent_Declare(imgui GIT_REPOSITORY https://github.com/ocornut/imgui.git GIT_TAG docking GIT_SHALLOW TRUE)
    FetchContent_GetProperties(imgui)
    if(NOT imgui_POPULATED)
        FetchContent_Populate(imgui)
    endif()
    FetchContent_Declare(imguizmo GIT_REPOSITORY https://github.com/CedricGuillemet/ImGuizmo.git GIT_TAG master GIT_SHALLOW TRUE)
    FetchContent_GetProperties(imguizmo)
    if(NOT imguizmo_POPULATED)
        FetchContent_Populate(imguizmo)
    endif()
    add_compile_definitions(MUK_USE_IMGUI)
endif()

include_directories(${CMAKE_SOURCE_DIR}/Engine)
add_subdirectory(Engine)
add_subdirectory(Runtime)
if(MUK_BUILD_EDITOR)
    add_subdirectory(Editor)
endif()

message(STATUS "Muk ${PROJECT_VERSION} Editor=${MUK_BUILD_EDITOR}")
